cmake_minimum_required(VERSION 3.22.1)
project(istanbulvice LANGUAGES CXX)
option(IV_BUILD_TESTS "Build host-side Part 6 smoke tests" OFF)

if(ANDROID)
  add_library(istanbulvice SHARED src/JniBridge.cpp src/Game.cpp src/Renderer.cpp src/Part8Systems.cpp)
  target_include_directories(istanbulvice PRIVATE include)
  target_compile_features(istanbulvice PRIVATE cxx_std_20)
  target_compile_options(istanbulvice PRIVATE -O3 -ffast-math -fno-exceptions -fno-rtti -Wall -Wextra)
  target_link_libraries(istanbulvice PRIVATE GLESv3 EGL android log)
elseif(IV_BUILD_TESTS)
  enable_testing()
  add_executable(iv_game_smoke src/Game.cpp src/Part8Systems.cpp tests/game_smoke.cpp)
  target_include_directories(iv_game_smoke PRIVATE include)
  target_compile_features(iv_game_smoke PRIVATE cxx_std_20)
  target_compile_options(iv_game_smoke PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME game_smoke COMMAND iv_game_smoke)
else()
  message(FATAL_ERROR "Use the Android NDK toolchain or configure with -DIV_BUILD_TESTS=ON for host smoke tests")
endif()
